# Записи DNS для почтового сервера
#
# Замените example.com, mail.example.com и 203.0.113.10 на свои значения.
# Имена указаны с точкой на конце — так их принимает большинство панелей.
# Точные значения для вашего домена печатает установщик.

# до установки 
# Нужна для выпуска сертификата Let's Encrypt: он выдаётся только тогда,
# когда имя уже указывает на сервер.
A       mail.example.com.       203.0.113.10

# после установки 
# Куда доставлять почту для домена.
# Если у домена уже есть MX на другой сервис — удалите старую запись.
MX      example.com.            mail.example.com.       приоритет 10

# Кому разрешено отправлять от имени домена.
# ~all — мягкий режим: письма с чужих серверов помечаются, но не отвергаются.
# После месяца наблюдений можно ужесточить до -all.
TXT     example.com.           "v=spf1 mx ~all"

# Отчёты о проверках. p=none — только собирать статистику.
# Через месяц можно перевести в p=quarantine.
TXT     _dmarc.example.com.    "v=DMARC1; p=none; rua=mailto:postmaster@example.com"

# Подпись писем. Значение выдаёт установщик, посмотреть заново:
#   cat /opt/mailserver/docker-data/dms/config/rspamd/dkim/*.public.key
# Вставлять одной сплошной строкой, без переносов. Если панель сама
# добавляет кавычки — свои не ставить. Проверьте, что сохранилось начало
# "v=DKIM1; k=rsa;": некоторые панели обрезают значение по точке с запятой.
TXT     mail._domainkey.example.com.    "v=DKIM1; k=rsa; p=MIIBIjANBgkq..."

# ── не в DNS домена ──────────────────────────────────────────────────────
# PTR живёт в обратной зоне IP и принадлежит хостеру — из панели домена
# туда не попасть. Запрос в поддержку:
#   «Прошу настроить PTR для 203.0.113.10 на mail.example.com».
# PTR   203.0.113.10       →  mail.example.com.

# ── проверка ─────────────────────────────────────────────────────────────
# dig +short MX example.com @1.1.1.1
# dig +short TXT example.com @1.1.1.1
# dig +short TXT _dmarc.example.com @1.1.1.1
# dig +short TXT mail._domainkey.example.com @1.1.1.1
# dig +short -x 203.0.113.10 @1.1.1.1
# dig +short mail.example.com @1.1.1.1
